cmake_minimum_required(VERSION 3.22)
project(fluent_cohere_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
if(NOT COHERE_SOURCE OR NOT COHERE_JSON_INCLUDE OR NOT COHERE_BUILD_ID)
  message(FATAL_ERROR "Use tool/build_cohere_asr.py with pinned inputs")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/bigobj>)
endif()
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_VULKAN OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_CUDA OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_X86_CONSERVATIVE ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(CMAKE_C_VISIBILITY_PRESET hidden)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
  add_compile_options(-include "${COHERE_SYMBOL_PREFIX_HEADER}")
else()
  set(TRANSCRIBE_BUILD_SHARED ON CACHE BOOL "" FORCE)
  set(TRANSCRIBE_GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "x64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  endif()
endif()
add_subdirectory("${COHERE_SOURCE}" "${CMAKE_BINARY_DIR}/transcribe" EXCLUDE_FROM_ALL)
# The upstream 0.4.1 target uses CMAKE_SOURCE_DIR for its public include path.
# Supply the real path when embedding it as this isolated subproject.
target_include_directories(transcribe PUBLIC "${COHERE_SOURCE}/include")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(fluent_cohere_asr STATIC cohere_worker.cpp)
  set(cohere_target fluent_cohere_asr)
elseif(ANDROID)
  add_library(fluent_cohere_asr SHARED cohere_worker.cpp cohere_jni.cpp)
  set(cohere_target fluent_cohere_asr)
  target_link_options(${cohere_target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
else()
  add_executable(cohere_asr_worker cohere_worker.cpp)
  set(cohere_target cohere_asr_worker)
  target_compile_definitions(${cohere_target} PRIVATE FLUENT_COHERE_DESKTOP=1)
endif()
target_include_directories(${cohere_target} PRIVATE "${COHERE_JSON_INCLUDE}")
target_link_libraries(${cohere_target} PRIVATE transcribe)
target_compile_definitions(${cohere_target} PRIVATE FLUENT_COHERE_BUILD_ID="${COHERE_BUILD_ID}")
set_target_properties(${cohere_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
