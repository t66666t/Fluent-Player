cmake_minimum_required(VERSION 3.24)
project(fluent_parakeet_ja LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  # All shipped private modules carry their CRT; no new user installation.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
if(NOT PARAKEET_SOURCE OR NOT PARAKEET_GGML_SOURCE OR NOT PARAKEET_JSON_INCLUDE)
  message(FATAL_ERROR "Use tool/build_parakeet_ja.py to prepare pinned inputs")
endif()
find_package(Threads REQUIRED)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  set(CMAKE_C_VISIBILITY_PRESET hidden)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
  add_compile_options(-include "${PARAKEET_SYMBOL_PREFIX_HEADER}")
else()
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "x64" OR (NOT CMAKE_GENERATOR_PLATFORM AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64"))
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  endif()
endif()
if(APPLE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
  add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/bigobj>)
endif()
add_subdirectory("${PARAKEET_GGML_SOURCE}" "${CMAKE_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
add_library(parakeet_audio_core STATIC
  "${PARAKEET_SOURCE}/src/core/mel.cpp"
  "${PARAKEET_SOURCE}/src/core/gguf_loader.cpp"
  "${PARAKEET_SOURCE}/src/crispasr_imatrix.cpp"
  "${PARAKEET_SOURCE}/src/parakeet.cpp"
  "${PARAKEET_SOURCE}/src/parakeet_orchestrate.cpp")
target_include_directories(parakeet_audio_core PUBLIC "${PARAKEET_SOURCE}/src" "${PARAKEET_SOURCE}/include")
target_include_directories(parakeet_audio_core PRIVATE "${PARAKEET_GGML_SOURCE}/src")
target_link_libraries(parakeet_audio_core PUBLIC ggml ggml-base Threads::Threads)
if(NOT GGML_BACKEND_DL)
  target_link_libraries(parakeet_audio_core PUBLIC ggml-cpu)
  foreach(backend metal vulkan cuda)
    if(TARGET ggml-${backend})
      target_link_libraries(parakeet_audio_core PUBLIC ggml-${backend})
    endif()
  endforeach()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(fluent_parakeet_ja STATIC parakeet_ja_worker.cpp)
  set(parakeet_target fluent_parakeet_ja)
elseif(ANDROID)
  add_library(fluent_parakeet_ja SHARED parakeet_ja_worker.cpp parakeet_ja_jni.cpp)
  set(parakeet_target fluent_parakeet_ja)
  target_link_options(${parakeet_target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
  add_executable(parakeet_ja_worker parakeet_ja_worker.cpp)
  set(parakeet_target parakeet_ja_worker)
  target_compile_definitions(${parakeet_target} PRIVATE FLUENT_PARAKEET_JA_DESKTOP=1)
endif()
target_include_directories(${parakeet_target} PRIVATE "${PARAKEET_JSON_INCLUDE}")
target_link_libraries(${parakeet_target} PRIVATE parakeet_audio_core)
target_compile_definitions(${parakeet_target} PRIVATE FLUENT_PARAKEET_JA_BUILD_ID="${PARAKEET_BUILD_ID}")
if(WIN32)
  target_link_libraries(${parakeet_target} PRIVATE ntdll bcrypt userenv ws2_32 advapi32)
elseif(APPLE)
  target_link_libraries(${parakeet_target} PRIVATE "-framework Security" "-framework SystemConfiguration")
elseif(UNIX)
  target_link_libraries(${parakeet_target} PRIVATE dl m)
endif()
if(APPLE)
  set(parakeet_rpath "@loader_path")
else()
  set(parakeet_rpath "$ORIGIN")
endif()
set_target_properties(${parakeet_target} PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "${parakeet_rpath}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
