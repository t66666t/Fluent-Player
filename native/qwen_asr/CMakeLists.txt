cmake_minimum_required(VERSION 3.24)
project(fluent_qwen_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  # All shipped private modules carry their CRT; no new user installation.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
if(NOT QWEN_SOURCE OR NOT QWEN_GGML_SOURCE OR NOT QWEN_TEXT_LIBRARY OR NOT QWEN_JSON_INCLUDE)
  message(FATAL_ERROR "Use tool/build_qwen_asr.py to prepare pinned inputs")
endif()
find_package(Threads REQUIRED)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  set(CMAKE_C_VISIBILITY_PRESET hidden)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
  add_compile_options(-include "${QWEN_SYMBOL_PREFIX_HEADER}")
else()
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "x64" OR (NOT CMAKE_GENERATOR_PLATFORM AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64"))
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  endif()
endif()
if(APPLE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
  add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/bigobj>)
endif()
add_subdirectory("${QWEN_GGML_SOURCE}" "${CMAKE_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
add_library(qwen_audio_core STATIC
  "${QWEN_SOURCE}/src/core/mel.cpp"
  "${QWEN_SOURCE}/src/core/gguf_loader.cpp"
  "${QWEN_SOURCE}/src/crispasr_imatrix.cpp"
  "${QWEN_SOURCE}/crisp_audio/src/audio_tower.cpp"
  "${QWEN_SOURCE}/src/qwen3_asr.cpp"
  "${QWEN_SOURCE}/src/ecapa_lid.cpp")
target_include_directories(qwen_audio_core PUBLIC "${QWEN_SOURCE}/src" "${QWEN_SOURCE}/crisp_audio/include")
target_include_directories(qwen_audio_core PRIVATE "${QWEN_GGML_SOURCE}/src")
target_link_libraries(qwen_audio_core PUBLIC ggml ggml-base Threads::Threads)
if(NOT GGML_BACKEND_DL)
  target_link_libraries(qwen_audio_core PUBLIC ggml-cpu)
  foreach(backend metal vulkan cuda)
    if(TARGET ggml-${backend})
      target_link_libraries(qwen_audio_core PUBLIC ggml-${backend})
    endif()
  endforeach()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(fluent_qwen_asr STATIC qwen_worker.cpp)
  set(qwen_target fluent_qwen_asr)
elseif(ANDROID)
  add_library(fluent_qwen_asr SHARED qwen_worker.cpp qwen_jni.cpp)
  set(qwen_target fluent_qwen_asr)
  target_link_options(${qwen_target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
  add_executable(qwen_asr_worker qwen_worker.cpp)
  set(qwen_target qwen_asr_worker)
  target_compile_definitions(${qwen_target} PRIVATE FLUENT_QWEN_DESKTOP=1)
endif()
target_include_directories(${qwen_target} PRIVATE "${QWEN_JSON_INCLUDE}")
target_link_libraries(${qwen_target} PRIVATE qwen_audio_core "${QWEN_TEXT_LIBRARY}")
target_compile_definitions(${qwen_target} PRIVATE FLUENT_QWEN_BUILD_ID="${QWEN_BUILD_ID}")
if(WIN32)
  target_link_libraries(${qwen_target} PRIVATE ntdll bcrypt userenv ws2_32 advapi32)
elseif(APPLE)
  target_link_libraries(${qwen_target} PRIVATE "-framework Security" "-framework SystemConfiguration")
elseif(UNIX)
  target_link_libraries(${qwen_target} PRIVATE dl m)
endif()
if(APPLE)
  set(qwen_rpath "@loader_path")
else()
  set(qwen_rpath "$ORIGIN")
endif()
set_target_properties(${qwen_target} PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "${qwen_rpath}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
