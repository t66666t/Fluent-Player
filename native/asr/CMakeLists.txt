cmake_minimum_required(VERSION 3.24)
project(fluent_asr_acceleration LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT SHERPA_SOURCE_DIR OR NOT ASR_ORT_INCLUDE OR NOT ASR_ORT_LIBRARY OR NOT ASR_JSON_INCLUDE)
  message(FATAL_ERROR "Provide pinned SHERPA_SOURCE_DIR, ASR_ORT_INCLUDE, ASR_ORT_LIBRARY and ASR_JSON_INCLUDE")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tool/prepare_asr_native.py"
  "${SHERPA_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}" "${ASR_JSON_INCLUDE}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
foreach(feature PYTHON JNI TTS SPEAKER_DIARIZATION PORTAUDIO WEBSOCKET BINARY TESTS)
  set(SHERPA_ONNX_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_DIRECTML OFF CACHE BOOL "" FORCE)
set(location_onnxruntime_header_dir "${ASR_ORT_INCLUDE}" CACHE PATH "" FORCE)
set(location_onnxruntime_lib "${ASR_ORT_LIBRARY}" CACHE FILEPATH "" FORCE)
add_subdirectory("${SHERPA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/sherpa" EXCLUDE_FROM_ALL)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  # Merge only sherpa and its static dependencies into the final archive.
  # OrtGetApiBase remains undefined and is resolved by OCR's existing ORT pod.
  add_library(fluent_asr_accel STATIC asr_accelerator.cpp)
  set(asr_target fluent_asr_accel)
elseif(ANDROID)
  add_library(fluent_asr_accel SHARED asr_accelerator.cpp)
  set(asr_target fluent_asr_accel)
else()
  add_executable(asr_worker asr_accelerator.cpp)
  set(asr_target asr_worker)
endif()
target_link_libraries(${asr_target} PRIVATE sherpa-onnx-c-api "${ASR_ORT_LIBRARY}")
target_include_directories(${asr_target} PRIVATE "${ASR_ORT_INCLUDE}")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_compile_definitions(${asr_target} PRIVATE FLUENT_ASR_IOS=1)
endif()
target_include_directories(${asr_target} PRIVATE "${SHERPA_SOURCE_DIR}")
set(asr_build_material "")
foreach(source asr_accelerator.cpp firered_ctc.h session_options.inc fluent_asr_observer.h mnn_session.h ../asr_mnn/mnn_graph.h)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND asr_build_material "${source_hash}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../tool/prepare_asr_native.py" source_hash)
string(SHA256 asr_build_id "${asr_build_material}${source_hash}")
target_compile_definitions(${asr_target} PRIVATE FLUENT_ASR_BUILD_ID="${asr_build_id}")
if(WIN32)
  target_compile_options(${asr_target} PRIVATE /utf-8)
  set_property(TARGET ${asr_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(${asr_target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${asr_target} PRIVATE dxgi d3d12)
elseif(ANDROID)
  target_link_libraries(${asr_target} PRIVATE log dl)
  target_link_options(${asr_target} PRIVATE "-Wl,-z,max-page-size=16384")
elseif(UNIX AND NOT APPLE)
  target_link_libraries(${asr_target} PRIVATE dl)
endif()
set_target_properties(${asr_target} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
  INSTALL_RPATH "$ORIGIN")
if(APPLE)
  set_target_properties(${asr_target} PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
install(TARGETS ${asr_target} RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION .)
