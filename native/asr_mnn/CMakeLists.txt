cmake_minimum_required(VERSION 3.24)
project(fluent_asr_mnn LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT OCR_MNN_SOURCE OR NOT OCR_ORT_INCLUDE)
  message(FATAL_ERROR "Provide pinned OCR_MNN_SOURCE and OCR_ORT_INCLUDE")
endif()
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_WIN_RUNTIME_MT ON CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER ON CACHE BOOL "" FORCE)
set(MNN_BUILD_PROTOBUFFER ON CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM OFF CACHE BOOL "" FORCE)
set(MNN_AVX512 OFF CACHE BOOL "" FORCE)
set(MNN_OPENMP OFF CACHE BOOL "" FORCE)
set(MNN_USE_SYSTEM_LIB OFF CACHE BOOL "" FORCE)
set(MNN_SUPPORT_INTEL_SUBGROUP OFF CACHE BOOL "" FORCE)
set(MNN_VULKAN_IMAGE OFF CACHE BOOL "" FORCE)
if(APPLE)
  set(MNN_METAL ON CACHE BOOL "" FORCE)
  set(MNN_METAL_REBUILD ON CACHE BOOL "" FORCE)
  set(MNN_METAL_OFFLINE_COMPILE OFF CACHE BOOL "" FORCE)
  set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
  set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
else()
  set(MNN_METAL OFF CACHE BOOL "" FORCE)
  set(MNN_OPENCL ON CACHE BOOL "" FORCE)
  set(MNN_VULKAN ON CACHE BOOL "" FORCE)
endif()
# Upstream's converter and backends resolve bundled headers against this root.
set(_ocr_root "${CMAKE_SOURCE_DIR}")
set(CMAKE_SOURCE_DIR "${OCR_MNN_SOURCE}")
add_subdirectory("${OCR_MNN_SOURCE}" "${CMAKE_BINARY_DIR}/mnn" EXCLUDE_FROM_ALL)
set(CMAKE_SOURCE_DIR "${_ocr_root}")
add_library(fluent_asr_mnn SHARED mnn_graph.cpp ../ocr/ocr_gpu_device.cpp)
target_compile_definitions(fluent_asr_mnn PRIVATE FLUENT_ASR_MNN_BUILD=1)
target_include_directories(fluent_asr_mnn PRIVATE "${OCR_MNN_SOURCE}/include"
  "${OCR_MNN_SOURCE}/source"
  "${OCR_MNN_SOURCE}/tools/converter/include" "${OCR_ORT_INCLUDE}")
if(MSVC)
  set_property(TARGET fluent_asr_mnn PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded$<$<CONFIG:Debug>:Debug>)
  target_compile_options(fluent_asr_mnn PRIVATE /utf-8)
  target_compile_definitions(fluent_asr_mnn PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_options(fluent_asr_mnn PRIVATE /WHOLEARCHIVE:$<TARGET_FILE:MNNConvertDeps>)
  target_link_options(fluent_asr_mnn PRIVATE /WHOLEARCHIVE:$<TARGET_FILE:MNN>)
elseif(APPLE)
  target_link_options(fluent_asr_mnn PRIVATE -Wl,-all_load)
else()
  # Vulkan kernels register through static initializers. Preserve them just as
  # upstream's static MNN executables do; otherwise a compiled backend vanishes.
  target_link_libraries(fluent_asr_mnn PRIVATE -Wl,--whole-archive MNNConvertDeps MNN -Wl,--no-whole-archive)
  target_link_libraries(fluent_asr_mnn PRIVATE ${CMAKE_DL_LIBS})
endif()
target_link_libraries(fluent_asr_mnn PRIVATE MNNConvertDeps MNN)
if(ANDROID)
  target_link_libraries(fluent_asr_mnn PRIVATE log)
  target_link_options(fluent_asr_mnn PRIVATE -Wl,-z,max-page-size=16384)
endif()
set_target_properties(fluent_asr_mnn PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON BUILD_WITH_INSTALL_RPATH TRUE
  INSTALL_RPATH "$ORIGIN" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(APPLE)
  set_target_properties(fluent_asr_mnn PROPERTIES INSTALL_RPATH "@loader_path")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(fluent_asr_mnn PROPERTIES FRAMEWORK TRUE
    FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER com.fluentplayer.asr-mnn
    OUTPUT_NAME FluentAsrMnn INSTALL_NAME_DIR "@rpath"
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
endif()
