cmake_minimum_required(VERSION 3.13)
project(app_network LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_library(app_network_plugin SHARED "app_network_plugin.cc")
apply_standard_settings(app_network_plugin)
target_compile_definitions(app_network_plugin PRIVATE FLUTTER_PLUGIN_IMPL)
target_include_directories(app_network_plugin INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(app_network_plugin PRIVATE flutter PkgConfig::GIO)
set(app_network_bundled_libraries "" PARENT_SCOPE)
