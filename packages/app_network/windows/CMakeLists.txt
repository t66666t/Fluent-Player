cmake_minimum_required(VERSION 3.14)
project(app_network LANGUAGES CXX)
add_library(app_network_plugin SHARED "app_network_plugin.cpp")
apply_standard_settings(app_network_plugin)
target_compile_features(app_network_plugin PRIVATE cxx_std_17)
target_compile_definitions(app_network_plugin PRIVATE FLUTTER_PLUGIN_IMPL NOMINMAX)
target_include_directories(app_network_plugin INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(app_network_plugin PRIVATE flutter flutter_wrapper_plugin winhttp iphlpapi ws2_32)
set(app_network_bundled_libraries "" PARENT_SCOPE)
